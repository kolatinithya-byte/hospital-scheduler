# How to Run this Java Project

git clone https://github.com/kolatinithya-byte/hospital-scheduler.git
cd hospital-scheduler/hospital-scheduler

# Import DB from /db folder
mysql -u root -p < db/schema.sql

# Configure DB in src/main/java/config/DatabaseConfig.java
mvn clean compile
mvn exec:java -Dexec.mainClass="com.hospital.Main"

# Server starts on http://localhost:8080
